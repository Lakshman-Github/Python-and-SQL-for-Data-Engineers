5,Divya,HR,50000

import pandas as pd

# Read CSV file
df = pd.read_csv("employees.csv")

# 1. Convert names to uppercase
df["name"] = df["name"].str.upper()

# 2. Multiplication
df["bonus"] = df["salary"] * 0.10

# 3. Calculate total salary
df["total_salary"] = df["salary"] + df["bonus"]

# 4. Rename column
df = df.rename(columns={"name": "employee_name"})

# 5. Filter employees with salary > 35000
df = df[df["salary"] > 35000]

# 6. Sort salary in descending order
df = df.sort_values(by="salary", ascending=False)

# 7. Save transformed data
df.to_csv("transformed_employees.csv", index=False)
