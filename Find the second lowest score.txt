n = int(input())

students = []
scores = []

for i in range(n):
    name = input()
    score = float(input())
    students.append([name, score])
    scores.append(score)
    
scores = sorted(set(scores))
second_lowest = scores[1]

for name, score in sorted(students):
    if score == second_lowest:
        print(name)
